// Numeric enum - values are auto-incremented from 0 or specified value
enum Color {
    Red = 1,    // 1
    Green,      // 2 (auto-incremented)
    Blue        // 3 (auto-incremented)
}

// Accessing VALUES (number) by KEY (string)
enum Color { Red = 1, Green = 2, Blue = 3 };

// Method 1: Access value using dot notation
let colorValue: Color = Color.Green;  // Returns: 2
console.log(colorValue);              // Output: 2

// Method 2: Direct access
console.log(Color.Red);    // Output: 1
console.log(Color.Blue);   // Output: 3

// Accessing KEYS (string) by VALUE (number)
enum Color { Red = 1, Green = 2, Blue = 3 };

// Using bracket notation with number
let colorName: string = Color[2];  // Returns: "Green"
console.log(colorName);            // Output: "Green"

console.log(Color[1]);  // Output: "Red"
console.log(Color[3]);  // Output: "Blue"

enum RequestType {
    GET = 1,
    POST = 2,
    PUT = 3,
    DELETE = 4
}

// Access values (numbers)
console.log(RequestType.GET);     // Output: 1
console.log(RequestType.POST);    // Output: 2

// Access keys (strings)
console.log(RequestType[1]);      // Output: "GET"
console.log(RequestType[2]);      // Output: "POST"

// Using enum in function parameters
function handleRequest(method: RequestType): void {
    console.log(`Handling ${RequestType[method]} request`);
}

handleRequest(RequestType.POST);  // Output: "Handling POST request"

# Class, Object, Constructor, Inheritance
#--------------------------------------------------------

# Basic class
class Person {
  name: string;
  age: number;
  
  constructor(name: string, age: number) {
    this.name = name;
    this.age = age;
  }
  
  greet(): void {
    console.log(`Hello, I'm ${this.name}`);
  }
}

# Create object
const person1 = new Person("John", 25);
person1.greet();

#--------------------------------------------------------
# Inheritance with super()
#--------------------------------------------------------
class Employee extends Person {
  salary: number;
  
  constructor(name: string, age: number, salary: number) {
    super(name, age);  // Call parent constructor
    this.salary = salary;
  }
  
  showSalary(): void {
    console.log(`${this.name}'s salary: $${this.salary}`);
  }
}

# Create employee object
const emp1 = new Employee("Alice", 30, 50000);
emp1.greet();
emp1.showSalary();

class BankAccount {
  public accountNo: string;    // Accessible anywhere
  private balance: number;     // Only within this class
  protected bankName: string;  // Within class and subclasses
  
  constructor(accountNo: string, balance: number, bankName: string) {
    this.accountNo = accountNo;
    this.balance = balance;
    this.bankName = bankName;
  }
  
  getBalance(): number {
    return this.balance;
  }
}

class Container<T> {
  private value: T;
  
  constructor(value: T) {
    this.value = value;
  }
  
  getValue(): T {
    return this.value;
  }
  
  setValue(newValue: T): void {
    this.value = newValue;
  }
}

# Generic with Constraint
#--------------------------------------------------------
interface Identifiable {
  id: number;
}

class Repository<T extends Identifiable> {
  private items: T[] = [];
  
  add(item: T): void {
    this.items.push(item);
  }
  
  findById(id: number): T | undefined {
    return this.items.find(item => item.id === id);
  }
}

# Use constrained generic
interface User extends Identifiable {
  name: string;
}

const userRepo = new Repository<User>();
userRepo.add({ id: 1, name: "John" });
console.log(userRepo.findById(1));

# Getter and Setter
#--------------------------------------------------------
class Temperature {
  private _celsius: number = 0;
  
  get celsius(): number {
    return this._celsius;
  }
  
  set celsius(value: number) {
    if (value < -273.15) {
      throw new Error("Temperature below absolute zero!");
    }
    this._celsius = value;
  }
  
  get fahrenheit(): number {
    return (this._celsius * 9/5) + 32;
  }
}

const temp = new Temperature();
temp.celsius = 25;  // Uses setter
console.log(temp.fahrenheit);  // Uses getter

# Complete Example: All Concepts Combined
#--------------------------------------------------------
abstract class Payment {
  abstract processPayment(amount: number): boolean;
}

interface Refundable {
  processRefund(): void;
}

class CreditCardPayment extends Payment implements Refundable {
  constructor(private cardNumber: string) {
    super();
  }
  
  processPayment(amount: number): boolean {
    console.log(`Processing $${amount} via credit card`);
    return true;
  }
  
  processRefund(): void {
    console.log("Processing refund to credit card");
  }
}

# Use all features
const payment = new CreditCardPayment("1234-5678-9012-3456");
payment.processPayment(100);
payment.processRefund();
